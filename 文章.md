### nativebridge_hook.py

```python
import frida, time, subprocess

ADB = r'"D:\MuMuPlayer\nx_main\adb.exe" -s 127.0.0.1:7555'
FRIDA_HOST = '127.0.0.1:8080'
PKG = 'com.yaodu.supplychain'
MAIN_ACT = '.modular.main.activity.SplashActivity'

jscode = """
console.log('loading NativeBridge hook...');

var libhoudini = Process.findModuleByName('libhoudini.so');
if (!libhoudini) {
    console.log('libhoudini not found');
}

var NativeBridgeItf = Module.findExportByName('libhoudini.so', 'NativeBridgeItf');
if (!NativeBridgeItf) {
    console.log('NativeBridgeItf not found');
} else {
    console.log('NativeBridgeItf @ ' + NativeBridgeItf);
    
    var loadLibrary = NativeBridgeItf.add(0x10).readPointer();
    var loadLibraryExt = NativeBridgeItf.add(0x70).readPointer();
    console.log('loadLibrary @ ' + loadLibrary);
    console.log('loadLibraryExt @ ' + loadLibraryExt);
    
    // hook loadLibrary
    if (!loadLibrary.isNull()) {
        Interceptor.attach(loadLibrary, {
            onEnter: function(args) {
                try {
                    var path = args[0].readCString();
                    console.log('[NB.loadLib] ' + path + ' flags=' + args[1].toInt32());
                } catch(e) {}
            },
            onLeave: function(retval) {
                console.log('[NB.loadLib] ret=' + retval);
            }
        });
        console.log('hook loadLibrary OK');
    }
    
    // hook loadLibraryExt  
    if (!loadLibraryExt.isNull()) {
        Interceptor.attach(loadLibraryExt, {
            onEnter: function(args) {
                try {
                    var path = args[0].readCString();
                    console.log('[NB.loadLibExt] ' + path + ' flags=' + args[1].toInt32());
                } catch(e) {}
            },
            onLeave: function(retval) {
                console.log('[NB.loadLibExt] ret=' + retval);
            }
        });
        console.log('hook loadLibraryExt OK');
    }
}
"""

def on_msg(msg, data):
    p = msg.get('payload', '')
    if p: print(p)

dev = frida.get_device_manager().add_remote_device(FRIDA_HOST)

# kill and restart app
subprocess.run(ADB + ' shell "su -c \'am force-stop ' + PKG + '; sleep 1\'"', shell=True)
subprocess.run(ADB + ' shell "su -c \'am start -n ' + PKG + '/' + MAIN_ACT + '\' 2>/dev/null"', shell=True)

# wait for PID
pid = None
for i in range(30):
    time.sleep(0.3)
    try:
        res = subprocess.check_output(ADB + ' shell "su -c pidof ' + PKG + '"', shell=True).decode().strip()
        if res:
            pid = int(res.split()[0])
            print('PID=' + str(pid) + ' @ ' + str((i+1)*0.3) + 's')
            break
    except: pass

if not pid:
    print('no PID')
else:
    try:
        session = dev.attach(pid)
        script = session.create_script(jscode)
        script.on('message', on_msg)
        script.load()
        print('capturing 30s...')
        time.sleep(30)
        session.detach()
    except Exception as e:
        print('err: ' + str(e))

print('done')
```

### hook_openmem.py 共鞥是精准抓取

```python
import frida, time, subprocess

ADB = '"D:\\MuMuPlayer\\nx_main\\adb.exe" -s 127.0.0.1:7555'
PKG = 'com.yaodu.supplychain'
MAIN = '.modular.main.activity.SplashActivity'

jscode = """
function hookAll() {
    ['libdexfile.so', 'libart.so', 'libdexfile.so'].forEach(function(modname) {
        var mod = Process.findModuleByName(modname);
        if (!mod) return;
        mod.enumerateExports().forEach(function(s) {
            if (s.name.indexOf('OpenMemory') >= 0 || s.name.indexOf('OpenCommon') >= 0) {
                try {
                    Interceptor.attach(s.address, {
                        onEnter: function(args) {
                            var base = args[0];
                            var size = args[1].toInt32();
                            send('CALL ' + s.name + ' base=' + base + ' size=' + size);
                            if (size > 1000 && size < 60000000 && !base.isNull()) {
                                try {
                                    var magic = base.readByteArray(8);
                                    var a = new Uint8Array(magic);
                                    send('  magic: ' + Array.from(a).map(function(b){return b.toString(16)}).join(' '));
                                    if (a[0]===0x64 && a[1]===0x65 && a[2]===0x78) {
                                        var buf = base.readByteArray(size);
                                        var fn = '/data/local/tmp/openmem_' + size + '.dex';
                                        var f = new File(fn, 'wb'); f.write(buf); f.close();
                                        send('  SAVED ' + fn);
                                    }
                                } catch(e) { send('  read err: ' + e); }
                            }
                        }
                    });
                    send('HOOKED ' + s.name);
                } catch(e) {}
            }
        });
    });
    send('all hooks installed');
}
hookAll();
"""

def on_msg(msg, data):
    p = msg.get('payload', '')
    if p: print(p)

dev = frida.get_device_manager().add_remote_device('127.0.0.1:8080')

subprocess.run(ADB + ' shell "su -c \'am force-stop ' + PKG + '; sleep 1\'"', shell=True)
subprocess.run(ADB + ' shell "su -c \'am start -n ' + PKG + '/' + MAIN + '\' 2>/dev/null"', shell=True)

pid = None
for i in range(60):
    time.sleep(0.1)
    try:
        res = subprocess.check_output(ADB + ' shell "su -c pidof ' + PKG + '"', shell=True).decode().strip()
        if res:
            pid = int(res.split()[0])
            print('PID=' + str(pid) + ' @ ' + str((i+1)*0.1) + 's')
            break
    except: pass

if pid:
    session = None
    time.sleep(1.0)
    for attempt in range(8):
        try:
            session = dev.attach(pid)
            print('attached on attempt ' + str(attempt+1))
            break
        except Exception as e:
            print('attach attempt ' + str(attempt+1) + ' failed: ' + str(e)[:60])
            time.sleep(0.5)
    if session:
        try:
            script = session.create_script(jscode)
            script.on('message', on_msg)
            script.load()
            print('hooks loaded, capturing 20s...')
            time.sleep(20)
            session.detach()
        except Exception as e:
            print('script failed: ' + str(e)[:100])
print('done')
```

### full_scan2.py 翻内存

```python
import frida, time, subprocess, sys

ADB = '"D:\\MuMuPlayer\\nx_main\\adb.exe" -s 127.0.0.1:7555'

jscode = """
var ranges = Process.enumerateRanges('r--');
ranges = ranges.concat(Process.enumerateRanges('rw-'));
ranges = ranges.concat(Process.enumerateRanges('r-x'));

var patterns = ['64 65 78 0a 30 33 35', '64 65 78 0a 30 33 39', '64 65 78 0a 30 33 37', '64 65 78 0a 30 33 36'];
var hits = [];
var idx = 0;

ranges.forEach(function(r) {
    if (r.size < 4096 || r.size > 60000000) return;
    patterns.forEach(function(pat) {
        try {
            var results = Memory.scanSync(r.base, r.size, pat);
            results.forEach(function(res) {
                var sz = res.address.add(0x20).readU32();
                if (sz > 1000 && sz < 60000000) {
                    try {
                        var buf = res.address.readByteArray(sz);
                        var fn = '/data/local/tmp/ds_' + idx + '_' + sz + '.dex';
                        var f = new File(fn, 'wb'); f.write(buf); f.close();
                        send('HIT ' + idx + ' @' + res.address + ' sz=' + sz + ' perm=' + r.protection + ' file=' + (r.file?r.file.path:'anon') + ' -> ' + fn);
                        idx++;
                    } catch(e2) {
                        send('HIT(no-save) @' + res.address + ' sz=' + sz + ' err=' + e2);
                    }
                }
            });
        } catch(e) {}
    });
});
send('DONE, saved ' + idx);
"""

def on_msg(msg, data):
    p = msg.get('payload', '')
    if p: print(p)

pid = int(sys.argv[1])
print(f'PID={pid}')
dev = frida.get_device_manager().add_remote_device('127.0.0.1:8080')
session = dev.attach(pid)
script = session.create_script(jscode)
script.on('message', on_msg)
script.load()
time.sleep(60)
session.detach()
print('done')
```
